export const splitLightDark = (value) => {
	// Splits light-dark(<light>, <dark>) at its top-level comma. Branches can hold nested
	// functions with their own commas, e.g. color-mix(in oklch, a, b).
	const args = value.match(/^light-dark\((.*)\)$/)?.[1];
	if (args === undefined) return undefined;
	let depth = 0;
	for (let i = 0; i < args.length; i++) {
		if (args[i] === '(') depth++;
		else if (args[i] === ')') depth--;
		else if (args[i] === ',' && depth === 0) {
			return { light: args.slice(0, i).trim(), dark: args.slice(i + 1).trim() };
		}
	}
	return undefined;
};

export const swatches = (token) => splitLightDark(token.value) ?? { all: token.value };

export const hueRow = (hue) => (
	<ColorItem key={hue.name} title={hue.name} subtitle={`--hue-${hue.name}-*`} colors={hue.shades} />
);

# Design tokens

Every custom property [suede-css](https://github.com/taurean/suede-css) declares on `:root`, read
from the installed package at build time, so this page follows a suede-css upgrade without edits.
Token values are human-owned: change them in suede-css, not here.

## Colour

### Surfaces

Semantic tokens. Components use these before reaching for a palette shade, because each one is a
`light-dark()` pair that follows the active colour scheme.

<ColorPalette>
	{tokens.surfaces.map((token) => (
		<ColorItem
			key={token.name}
			title={`--${token.name}`}
			subtitle={token.value}
			colors={swatches(token)}
		/>
	))}
</ColorPalette>

### Palette

Each hue has 11 OKLCH shades, `50` (lightest) to `950` (darkest), referenced as
`--hue-<name>-<shade>`. Hover a swatch for its value.

<style>{`
	/* ColorItem prints each value under its swatch. OKLCH strings are too long for 11 columns
	   and overlap. The swatch keeps the value as its title tooltip, so hide the printed copy. */
	.token-palette [title] > div > span { display: none; }

	/* Storybook's docs sheet stays light, but under a dark OS scheme suede-css sets body text to
	   a light colour, so anything inheriting it disappears. Pin this page's inheriting elements. */
	.token-palette > summary, .token-typeset { color: var(--hue-neutral-950); }
`}</style>

<details open className="token-palette">
	<summary>Chromatic ({tokens.chromaticHues.length} hues)</summary>
	<ColorPalette>{tokens.chromaticHues.map(hueRow)}</ColorPalette>
</details>

<details className="token-palette">
	<summary>Neutrals ({tokens.neutralHues.length} hues)</summary>
	<ColorPalette>{tokens.neutralHues.map(hueRow)}</ColorPalette>
</details>
